# 节点管理动态表

> 依赖 `Dataview` 插件，并开启 `JavaScript Queries`。
> 这是动态版：每次打开都会实时扫描 Vault，不依赖静态导出。

## 规则

- 扫描范围：仅扫描 `个人资料/服务器资料/` 下的 Markdown 文件，并排除 `节点管理/` 输出目录，避免自引用循环。
- 协议范围：当前只识别 `vless://` 与 `ss://`。
- 节点抽取：按行提取链接，支持普通文本与表格单元格；会清理 HTML 标签和行尾分隔符噪声。
- 节点命名：优先取链接 `#` 后名称（含 URL 解码）；无名称时回退为 `文件名-序号`。
- 节点过滤：会过滤协议名/临时标记名/明显噪声命名，只保留有效节点名。
- 去重统计：按 `协议 + 链接主体(忽略#后缀)` 计算 `重复数`。
- 属性判定：`IPLC/IEPL` -> `专线`；`FACHOST/Uzumaru/Hinet/Frontier` -> `家宽`；`速维云/无忧云/YT` -> `IX`（其余留空）。
- 状态默认值：`不同步`。

```dataviewjs
const OUTPUT_FOLDER = "节点管理/";
const SOURCE_FOLDER = "个人资料/服务器资料/";
const SUPPORTED_PROTOCOLS = ["vless", "ss"];

function safeDecode(value) {
  try {
    return decodeURIComponent(value);
  } catch {
    return value;
  }
}

function getVendorName(title) {
  title = title.trim();
  const dashIndex = title.indexOf("-");
  const spaceIndex = title.indexOf(" ");
  const candidates = [dashIndex, spaceIndex].filter((index) => index > 0);
  if (!candidates.length) return title;
  return title.slice(0, Math.min(...candidates)).trim() || title;
}

function extractProtocol(url) {
  const match = url.match(/^([a-zA-Z0-9+.-]+):\/\//);
  return match ? match[1].toLowerCase() : "";
}

function extractNodeName(url, fallback) {
  const index = url.indexOf("#");
  if (index === -1) return fallback;
  let fragment = url.slice(index + 1).trim();
  fragment = safeDecode(fragment);
  fragment = fragment.replace(/\|/g, " ").replace(/\s+/g, " ").trim();
  return fragment || fallback;
}

function normalizeServerName(name) {
  let value = (name || "").trim();
  if (!value) return value;
  value = value.replace(/-\d{1,3}(?:\.\d{1,3}){3}-[A-Za-z0-9.-]+(?:-\d+)?$/i, "");
  value = value.replace(/-[A-Za-z]?\d{2,}\.[A-Za-z0-9.-]+(?:-\d+)?$/i, "");
  value = value.replace(/\s*-\s*\d{1,3}(?:\.\d{1,3}){2,3}$/i, "");
  value = value.replace(/\s*-\s*\d{1,3}(?:\.\d{1,3}){1,2}$/i, "");
  value = value.replace(/\s*-\s*[A-Za-z0-9.-]+\.[A-Za-z]{2,}(?:-\d+)?$/i, "");
  value = value.replace(/\s*-\s*[a-f0-9]{8,}$/i, "");
  return value.replace(/[-\s]+$/g, "").trim();
}

function buildServerNameFromTitle(title) {
  let value = title.trim();
  value = value.replace(/\s*-\s*\d{1,3}(?:\.\d{1,3}){3}.*$/i, "");
  value = value.replace(/\s+-\s+[A-Za-z0-9.-]+\.[A-Za-z]{2,}.*$/i, "");
  value = value.replace(/\s{2,}/g, " ").trim();
  return normalizeServerName(value);
}

function isMeaningfulNodeName(name) {
  const normalized = (name || "").trim().toLowerCase();
  if (!normalized) return false;
  if (SUPPORTED_PROTOCOLS.includes(normalized)) return false;
  if (["reality", "hy2", "hysteria", "hysteria2", "tuic", "trojan"].includes(normalized)) return false;
  if (["reality-", "ss-", "vmess-", "vless-"].some((prefix) => normalized.startsWith(prefix))) return false;
  if (/^[a-z]{2,5}-vless(?:[_+])reality-\d+$/.test(normalized)) return false;
  return normalized.length > 2;
}

function extractNodesFromText(text) {
  const protocolRegex = new RegExp(`(?<![A-Za-z0-9+.-])(${SUPPORTED_PROTOCOLS.join("|")}):\\/\\/`, "i");
  const nodes = [];
  for (const rawLine of text.split(/\r?\n/)) {
    const line = rawLine.trim();
    if (!line) continue;

    let workingLine = line;
    if (line.includes("|")) {
      for (const cell of line.split("|").map((part) => part.trim())) {
        if (protocolRegex.test(cell)) {
          workingLine = cell;
          break;
        }
      }
    }

    workingLine = workingLine.replace(/<[^>]+>/g, "");
    const match = workingLine.match(protocolRegex);
    if (!match || match.index == null) continue;

    let candidate = workingLine.slice(match.index).trim();
    candidate = candidate.replace(/^`+|`+$/g, "");
    candidate = candidate.replace(/^"+|"+$/g, "");
    candidate = candidate.replace(/^'+|'+$/g, "");
    candidate = candidate.replace(/\s+\|\s*$/g, "");
    candidate = candidate.replace(/\|\s*$/g, "");
    candidate = candidate.trim();
    if (candidate) nodes.push(candidate);
  }
  return nodes;
}

function normalizeUrlForDedup(url) {
  return url.replace(/#.*$/, "").trim();
}

function deriveAttribute(serverName, nodeName) {
  const haystack = `${serverName} ${nodeName}`.toLowerCase();
  if (haystack.includes("iplc") || haystack.includes("iepl") || haystack.includes("ipel")) return "专线";
  if (["fachost", "uzumaru", "hinet", "frontier"].some((keyword) => haystack.includes(keyword))) {
    return "家宽";
  }
  // global 只对 Lazycats / Uzumaru 算家宽
  if (haystack.includes("global") && (haystack.includes("lazycats") || haystack.includes("uzumaru"))) {
    return "家宽";
  }
  if (haystack.includes("速维云") || haystack.includes("无忧云") || haystack.includes("yt")) return "IX";
  return "";
}

function pickCanonicalServerName(items) {
  const candidates = items
    .map((item) => normalizeServerName(item.serverName))
    .filter(Boolean);
  if (!candidates.length) return "";

  const counts = new Map();
  for (const candidate of candidates) {
    counts.set(candidate, (counts.get(candidate) || 0) + 1);
  }

  return [...counts.keys()].sort((a, b) => {
    const scoreA = [counts.get(a), a.includes("…") || a.includes("...") ? 0 : 1, a.length];
    const scoreB = [counts.get(b), b.includes("…") || b.includes("...") ? 0 : 1, b.length];
    for (let i = 0; i < scoreA.length; i++) {
      if (scoreA[i] !== scoreB[i]) return scoreB[i] - scoreA[i];
    }
    return a.localeCompare(b, "zh-Hans");
  })[0];
}

const files = app.vault
  .getMarkdownFiles()
  .filter((file) => file.path.startsWith(SOURCE_FOLDER) && !file.path.startsWith(OUTPUT_FOLDER));
const rawRows = [];

for (const file of files) {
  const text = await app.vault.cachedRead(file);
  const urls = extractNodesFromText(text);
  if (!urls.length) continue;

  const vendorName = getVendorName(file.basename);
  const serverName = buildServerNameFromTitle(file.basename) || vendorName;
  let sequence = 0;

  for (const url of urls) {
    sequence += 1;
    const protocol = extractProtocol(url);
    const nodeName = extractNodeName(url, `${file.basename}-${sequence}`);
    if (!isMeaningfulNodeName(nodeName)) continue;
    const dedupBasis = normalizeUrlForDedup(url);
    rawRows.push({
      vendorName,
      serverName,
      nodeName,
      url,
      status: "不同步",
      notePath: file.path,
      duplicateKey: `${protocol}::${dedupBasis}`,
    });
  }
}

const rowsByNote = new Map();
for (const row of rawRows) {
  if (!rowsByNote.has(row.notePath)) rowsByNote.set(row.notePath, []);
  rowsByNote.get(row.notePath).push(row);
}

for (const items of rowsByNote.values()) {
  const canonical = pickCanonicalServerName(items);
  for (const item of items) {
    item.serverName = canonical || item.serverName;
    item.attribute = deriveAttribute(item.serverName, item.nodeName);
  }
}

const duplicateCounts = new Map();
for (const row of rawRows) {
  duplicateCounts.set(row.duplicateKey, (duplicateCounts.get(row.duplicateKey) || 0) + 1);
}

const rows = rawRows
  .map((row, index) => ({
    index: index + 1,
    ...row,
    duplicateCount: duplicateCounts.get(row.duplicateKey) || 1,
  }))
  .sort((a, b) => {
    return (
      a.vendorName.localeCompare(b.vendorName, "zh-Hans") ||
      a.serverName.localeCompare(b.serverName, "zh-Hans") ||
      a.nodeName.localeCompare(b.nodeName, "zh-Hans") ||
      a.notePath.localeCompare(b.notePath, "zh-Hans")
    );
  });

dv.table(
  ["序号", "VPS供应商名称", "VPS服务器名称", "节点信息", "状态", "属性", "重复数", "笔记"],
  rows.map((row, index) => [
    index + 1,
    row.vendorName,
    row.serverName,
    row.nodeName,
    row.status,
    row.attribute,
    row.duplicateCount,
    dv.fileLink(row.notePath),
  ])
);
```
