# RealmFlow 数据模型

## 核心数据模型

### User (用户)

```go
type User struct {
    ID           uint          `json:"id"`
    Username     string        `json:"username"`
    PasswordHash string        `json:"-"`
    TelegramID   string        `json:"telegram_id"`   // TG 绑定
    Role         string        `json:"role"`          // admin/user
    Quota        int64         `json:"quota"`         // 流量配额(字节)
    UsedTraffic  int64         `json:"used_traffic"`  // 已用流量
    NotifyConfig NotifyConfig  `json:"notify_config"` // 通知配置
    Status       string        `json:"status"`        // active/disabled
    CreatedAt    time.Time     `json:"created_at"`
    UpdatedAt    time.Time     `json:"updated_at"`
}
```

### NotifyConfig (通知配置)

```go
type NotifyConfig struct {
    Enabled     bool     `json:"enabled"`
    Events      []string `json:"events"`     // traffic_alert, node_down, etc.
    TrafficThreshold int `json:"traffic_threshold"` // 流量告警百分比
}
```

### Node (节点)

```go
type Node struct {
    ID        uint      `json:"id"`
    Name      string    `json:"name"`
    Host      string    `json:"host"`
    Port      int       `json:"port"`
    APIKey    string    `json:"api_key"`
    Status    string    `json:"status"`     // online/offline
    LastSeen  time.Time `json:"last_seen"`
    CreatedAt time.Time `json:"created_at"`
}
```

### Tunnel (隧道)

```go
type Tunnel struct {
    ID          uint      `json:"id"`
    UserID      uint      `json:"user_id"`
    NodeID      uint      `json:"node_id"`
    Name        string    `json:"name"`
    RemoteAddr  string    `json:"remote_addr"`
    RemotePort  int       `json:"remote_port"`
    Quota       int64     `json:"quota"`
    UsedTraffic int64     `json:"used_traffic"`
    DefaultEngine string  `json:"default_engine"` // nftables/realm - 默认引擎
    Status      string    `json:"status"`     // active/disabled
    CreatedAt   time.Time `json:"created_at"`
}
```

### Forward (转发)

```go
type Forward struct {
    ID         uint      `json:"id"`
    TunnelID   uint      `json:"tunnel_id"`
    Name       string    `json:"name"`
    ListenPort int       `json:"listen_port"`
    ListenIP   string    `json:"listen_ip"`
    TargetAddr string    `json:"target_addr"`
    TargetPort int       `json:"target_port"`
    Protocol   string    `json:"protocol"`    // tcp/udp
    Engine     string    `json:"engine"`      // nftables/realm - 转发引擎
    Enabled    bool      `json:"enabled"`
    Status     string    `json:"status"`      // running/stopped
    CreatedAt  time.Time `json:"created_at"`
}
```

### TrafficRecord (流量记录)

```go
type TrafficRecord struct {
    ID        uint      `json:"id"`
    ForwardID uint      `json:"forward_id"`
    TunnelID  uint      `json:"tunnel_id"`
    UserID    uint      `json:"user_id"`
    NodeID    uint      `json:"node_id"`
    Upload    int64     `json:"upload"`      // 上传字节数
    Download  int64     `json:"download"`    // 下载字节数
    Timestamp time.Time `json:"timestamp"`
}
```

### ServerStatus (服务器状态)

```go
type ServerStatus struct {
    ID         uint      `json:"id"`
    NodeID     uint      `json:"node_id"`
    CPU        float64   `json:"cpu"`         // CPU 使用率
    Memory     float64   `json:"memory"`      // 内存使用率
    Disk       float64   `json:"disk"`        // 磁盘使用率
    NetworkIn  int64     `json:"network_in"`  // 网络入流量
    NetworkOut int64     `json:"network_out"` // 网络出流量
    Timestamp  time.Time `json:"timestamp"`
}
```

### NotificationLog (通知日志)

```go
type NotificationLog struct {
    ID        uint      `json:"id"`
    UserID    uint      `json:"user_id"`
    Type      string    `json:"type"`        // traffic_alert, node_down
    Message   string    `json:"message"`
    SentAt    time.Time `json:"sent_at"`
    Status    string    `json:"status"`      // sent/failed
    Error     string    `json:"error,omitempty"`
}
```

## 索引设计

```sql
-- 用户表
CREATE UNIQUE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_telegram_id ON users(telegram_id);

-- 节点表
CREATE UNIQUE INDEX idx_nodes_name ON nodes(name);
CREATE INDEX idx_nodes_status ON nodes(status);

-- 隧道表
CREATE INDEX idx_tunnels_user_id ON tunnels(user_id);
CREATE INDEX idx_tunnels_node_id ON tunnels(node_id);
CREATE INDEX idx_tunnels_status ON tunnels(status);

-- 转发表
CREATE INDEX idx_forwards_tunnel_id ON forwards(tunnel_id);
CREATE INDEX idx_forwards_listen_port ON forwards(listen_port);
CREATE INDEX idx_forwards_enabled ON forwards(enabled);

-- 流量记录表
CREATE INDEX idx_traffic_records_forward_id ON traffic_records(forward_id);
CREATE INDEX idx_traffic_records_tunnel_id ON traffic_records(tunnel_id);
CREATE INDEX idx_traffic_records_user_id ON traffic_records(user_id);
CREATE INDEX idx_traffic_records_node_id ON traffic_records(node_id);
CREATE INDEX idx_traffic_records_timestamp ON traffic_records(timestamp);

-- 服务器状态表
CREATE INDEX idx_server_status_node_id ON server_status(node_id);
CREATE INDEX idx_server_status_timestamp ON server_status(timestamp);

-- 通知日志表
CREATE INDEX idx_notification_logs_user_id ON notification_logs(user_id);
CREATE INDEX idx_notification_logs_type ON notification_logs(type);
CREATE INDEX idx_notification_logs_sent_at ON notification_logs(sent_at);
```

## 数据保留策略

| 表 | 保留策略 |
|---|---|
| traffic_records | 保留 90 天 |
| server_status | 保留 30 天 |
| notification_logs | 保留 180 天 |
| 其他表 | 永久保留 |
